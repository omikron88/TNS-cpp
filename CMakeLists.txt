cmake_minimum_required(VERSION 3.16)
project(TNS VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TNS_STATIC "Link SDL and compiler runtimes statically (OS libraries remain dynamic)" OFF)
option(TNS_BUNDLED_SDL "Build pinned SDL2 source instead of system SDL2" OFF)
option(TNS_BUILD_GUI "Build SDL frontend" ON)
option(TNS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(cmake/Embed.cmake)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
tns_embed("${CMAKE_CURRENT_BINARY_DIR}/generated/roms.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/roms/bootGC.bin" "${CMAKE_CURRENT_SOURCE_DIR}/roms/bootSC.bin"
  "${CMAKE_CURRENT_SOURCE_DIR}/roms/chars.bin" "${CMAKE_CURRENT_SOURCE_DIR}/roms/charsITK.bin")
add_library(tns_core INTERFACE)
target_include_directories(tns_core INTERFACE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  # Java wrapping arithmetic: retain defined two's complement semantics in the port.
  target_compile_options(tns_core INTERFACE -Wall -Wextra -Wno-parentheses -fwrapv)
  if(TNS_SANITIZERS)
    target_compile_options(tns_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tns_core INTERFACE -fsanitize=address,undefined)
  endif()
endif()
if(TNS_BUILD_GUI)
  if(TNS_BUNDLED_SDL)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    # TNS has no audio device. Avoid optional third-party runtime dependencies.
    set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
    set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
    set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
    set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
    set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
    set(SDL_WAYLAND OFF CACHE BOOL "" FORCE)
    set(SDL_PIPEWIRE OFF CACHE BOOL "" FORCE)
    set(SDL_PULSEAUDIO OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
      GIT_TAG 5d249570393f7a37e037abf22cd6012a4cc56a71) # release-2.32.10
    FetchContent_MakeAvailable(SDL2)
  else()
    find_package(SDL2 2.0.10 CONFIG REQUIRED)
  endif()
  add_executable(tns src/main.cpp)
  add_library(tns_sdl INTERFACE)
  target_link_libraries(tns PRIVATE tns_core tns_sdl)
  if(TARGET SDL2::SDL2main)
    target_link_libraries(tns_sdl INTERFACE SDL2::SDL2main)
  endif()
  if(TNS_STATIC)
    if(NOT TARGET SDL2::SDL2-static)
      message(FATAL_ERROR "Static SDL2 unavailable. Set TNS_BUNDLED_SDL=ON or install an SDL2 static development package.")
    endif()
    target_link_libraries(tns_sdl INTERFACE SDL2::SDL2-static)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT APPLE)
      target_link_options(tns_sdl INTERFACE -static-libgcc -static-libstdc++)
      if(MINGW)
        target_link_options(tns_sdl INTERFACE -static)
      endif()
    endif()
  else()
    target_link_libraries(tns_sdl INTERFACE SDL2::SDL2)
  endif()
  install(TARGETS tns RUNTIME DESTINATION bin)
  install(FILES LICENSE THIRD_PARTY.md README.md DESTINATION share/doc/tns)
  install(FILES licenses/SDL2.txt DESTINATION share/doc/tns/licenses)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tns_tests tests/core_tests.cpp)
  target_link_libraries(tns_tests PRIVATE tns_core)
  add_test(NAME core COMMAND tns_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests")
  if(TNS_BUILD_GUI)
    add_executable(tns_ui_tests tests/ui_tests.cpp)
    target_link_libraries(tns_ui_tests PRIVATE tns_core tns_sdl)
    add_test(NAME sdl_ui COMMAND tns_ui_tests)
    set_tests_properties(sdl_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 30)
    add_test(NAME sdl_boot_gc COMMAND tns --frames 100 --screenshot gc.ppm --no-save)
    add_test(NAME sdl_boot_itk COMMAND tns --itk --frames 100 --screenshot itk.ppm --no-save)
    add_test(NAME external_roms COMMAND tns --headless --gc --frames 100 --rom-dir "${CMAKE_CURRENT_SOURCE_DIR}/roms" --no-save)
    set_tests_properties(sdl_boot_gc sdl_boot_itk PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 30)
  endif()
endif()
